```bash
# laravel_reached <name> <method> <path>
# Proves a request was answered by Laravel and not by the static frontend
# fallback: the static frontend never sets a cookie, and its fallback is a 404.
# The concrete status (419 for an unauthenticated POST without a CSRF token) is
# reported as evidence but is not required.
laravel_reached() {
    local name="$1" method="$2" path="$3"
    local headers code
    headers="$(curl -s -X "$method" -D - -o /dev/null "${BASE}${path}" | tr -d '\r')"
    code="$(printf '%s' "$headers" | awk 'NR==1{print $2}')"
    if [ "$code" != "404" ] && contains "${headers,,}" "set-cookie:"; then
        green "PASS  ${name} (${method} ${path} -> ${code}, answered by Laravel)"
        PASS=$((PASS + 1))
    else
        red "FAIL  ${name} (${method} ${path} -> ${code}, not answered by Laravel)"
        FAIL=$((FAIL + 1))
    fi
}

# livewire_runtime: discovers the content-addressed Livewire prefix from the
# real login page instead of copying a hash, then checks the runtime script and
# the technical endpoints behind it.
livewire_runtime() {
    local html script prefix headers body code
    html="$(curl -s "${BASE}/admin/login")"
    script="$(printf '%s' "$html" | grep -oE '/livewire-[^/"]+/livewire(\.min)?\.js(\?[^"]*)?' | head -n 1)"
    script="${script//&amp;/&}"
    if [ -z "$script" ]; then
        red "FAIL  livewire script discovered (/admin/login has no Livewire script)"
        FAIL=$((FAIL + 1))
        return
    fi
    green "PASS  livewire script discovered (${script})"
    PASS=$((PASS + 1))

    headers="$(curl -s -D - -o /tmp/portfolio-smoke-livewire.js "${BASE}${script}" | tr -d '\r')"
    body="$(head -c 2048 /tmp/portfolio-smoke-livewire.js)"
    rm -f /tmp/portfolio-smoke-livewire.js
    code="$(printf '%s' "$headers" | awk 'NR==1{print $2}')"
    if [ "$code" = "200" ] && contains "${headers,,}" "javascript" && ! contains "$body" 'Página no encontrada' && ! contains "$body" '<html'; then
        green "PASS  livewire runtime script (${script} -> 200 javascript)"
        PASS=$((PASS + 1))
    else
        red "FAIL  livewire runtime script (${script} -> ${code}, expected 200 javascript, not the frontend)"
        FAIL=$((FAIL + 1))
    fi

    prefix="$(printf '%s' "$script" | sed -E 's#^(/livewire-[^/]+/).*#\1#')"
    laravel_reached "livewire update reaches Laravel"      POST "${prefix}update"
    laravel_reached "livewire upload reaches Laravel"      POST "${prefix}upload-file"
}
```

```php
<?php

namespace App\Support;

use Closure;

/**
 * Refuses a console process that would create persistent media the web
 * runtime cannot use.
 *
 * `docker compose exec` runs as root by default. The local disk creates
 * private directories as 0700 and files as 0600 owned by the creating
 * process, so anything written by root is unreadable and unwritable for
 * Apache/PHP (www-data). The check is identity-based on purpose:
 * is_writable() is true for root and would not detect the problem.
 *
 * Every identity is resolved inside the container (the www-data account and
 * the owner of the media roots); nothing depends on a host UID.
 */
final class PersistentMediaWriterGuard
{
    public const RUNTIME_USER = 'www-data';

    private readonly Closure $effectiveUid;

    private readonly Closure $uidOfUser;

    private readonly Closure $ownerOf;

    /** @var list<string>|null */
    private readonly ?array $roots;

    /** @param list<string>|null $roots */
    public function __construct(?Closure $effectiveUid = null, ?Closure $uidOfUser = null, ?Closure $ownerOf = null, ?array $roots = null)
    {
        $this->effectiveUid = $effectiveUid ?? static fn (): int => posix_geteuid();
        $this->uidOfUser = $uidOfUser ?? static function (string $name): ?int {
            $entry = posix_getpwnam($name);

            return $entry === false ? null : (int) $entry['uid'];
        };
        $this->ownerOf = $ownerOf ?? static function (string $path): ?int {
            if (! is_dir($path)) {
                return null;
            }
            $owner = fileowner($path);

            return $owner === false ? null : $owner;
        };
        $this->roots = $roots;
    }

    public function assertRuntimeUser(string $command): void
    {
        $euid = ($this->effectiveUid)();
        $runtimeUid = ($this->uidOfUser)(self::RUNTIME_USER);

        $accepted = $euid !== 0 && $runtimeUid !== null && $euid === $runtimeUid;
        foreach ($this->roots() as $root) {
            $accepted = $accepted && ($this->ownerOf)($root) === $euid;
        }

        if (! $accepted) {
            throw new PersistentMediaWriterRejected(
                'This command writes persistent media and must be executed as the runtime user ('.self::RUNTIME_USER.'). '
                ."Run it with `docker compose exec -T -u ".self::RUNTIME_USER." api php artisan {$command}`."
            );
        }
    }

    /** @return list<string> */
    private function roots(): array
    {
        return $this->roots ?? [
            (string) config('filesystems.disks.local.root'),
            (string) config('filesystems.disks.public.root'),
        ];
    }
}
```
